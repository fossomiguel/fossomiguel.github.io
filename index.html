Trapeze.java
// Trapèze rectangle : deux bases, la hauteur (côté perpendiculaire aux bases)
// et le côté oblique.
public class Trapeze {
    private double grandeBase;
    private double petiteBase;
    private double hauteur;
    private double cote;

    // constructeurs
    public Trapeze() {
        grandeBase = 0;
        petiteBase = 0;
        hauteur = 0;
        cote = 0;
    }

    public Trapeze(double grandeBase, double petiteBase, double hauteur, double cote) {
        this.grandeBase = grandeBase;
        this.petiteBase = petiteBase;
        this.hauteur = hauteur;
        this.cote = cote;
    }

    // périmètre = somme des quatre côtés
    public double perimetre() {
        return grandeBase + petiteBase + hauteur + cote;
    }

    // aire = (grande base + petite base) x hauteur / 2
    public double aire() {
        return (grandeBase + petiteBase) * hauteur / 2;
    }

    // méthode d'affichage
    public void affichage() {
        System.out.println("Trapèze : grande base = " + grandeBase
                + ", petite base = " + petiteBase
                + ", hauteur = " + hauteur
                + ", côté = " + cote);
    }
}
Test.java
public class Test {
    public static void main(String[] args) {
        // création d'objets
        Trapeze t1 = new Trapeze();
        Trapeze t2 = new Trapeze(4, 2, 6, 8);

        // calcul du périmètre
        double p = t2.perimetre();
        System.out.println("perimetre = " + p);
        System.out.println("aire = " + t2.aire());
        t2.affichage();

        // le trapèze par défaut
        System.out.println("perimetre de t1 = " + t1.perimetre());

        // test de la classe Point
        Point origine = new Point();
        Point a = new Point(3, 5);
        origine.affichage();
        a.affichage();
    }
}
